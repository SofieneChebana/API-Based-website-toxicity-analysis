TABLES SCHEMA :

CREATE TABLE ALL_URL_SAFE(ID integer primary key,URL TEXT NOT NULL,SCORE INTEGER NOT NULL,UNIQUE(URL));
CREATE TABLE ALL_URL_NOT_SAFE(ID integer primary key,URL TEXT NOT NULL,SCORE INTEGER NOT NULL,UNIQUE(URL));
CREATE TABLE ALL_URL(ID integer primary key,URL TEXT NOT NULL,UNIQUE(URL));


CREATE TABLE ALL_URL_WITH_PAGE_RANK(
    ID INTEGER PRIMARY KEY,
    URL TEXT NOT NULL,
    PAGERANK REAL,
    LINK_FARM INTEGER,
    UNIQUE(URL)
);
Create table ALL_URL_FINAL(
	ID integer primary key,
	URL TEXT NOT NULL,
	NBR_API INTEGER NOT NULL,
	ORIGINS_API REAL,
	COUNTRY INTEGER,
	WEBCONTENT REAL,
	TOXIC INTEGER,
	SCORE REAL NOT NULL
);
